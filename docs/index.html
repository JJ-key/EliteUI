About

A cross-platform UI library for Roblox — PC, Mobile & Tablet

EliteUI is a modern UI library built for Roblox scripts. It features a sharp purple theme, smooth animations, and a simple API that works seamlessly on PC keyboards, mobile touchscreens, and tablets.

PC Support

Full keyboard navigation. RightShift toggles the UI. Drag-to-move window.

Mobile & Touch

Hold 3 fingers for 0.8 s to toggle. Touch-friendly 44 px tap targets throughout.

Purple & Black Theme

Sleek dark design with a customisable accent colour and smooth tweened animations.

Settings Persistence

JSON-based settings saving and a built-in Default Settings Tab. No extra setup.

Tabs & Sub-tabs

Organise your features with nested tabs and auto-generated section headers.

Simple API

One-line loadstring loader. Methods for buttons, toggles, sliders, dropdowns, and more.

Installation

Components

Interactive UI elements with live previews and copy-ready Lua code

Window

The main container. Creates a draggable window with tabs. Supports an optional Key System, Snow Effect, and per-device UI scaling.

Lua
local Window = EliteUI:CreateWindow({
    Name = "My Script Hub",
    LoadingTitle = "Loading...",
    Subtitle = "v1.0",
    ToggleKey = Enum.KeyCode.RightShift,

    UIScale = { PC = 1.0, Mobile = 0.85, Tablet = 0.9 },

    KeySystem = {
        Enabled = true,
        URL = "https://pastebin.com/raw/YOURKEY",
        Title = "Key Required",
        Subtitle = "Enter your key to continue",
        SaveKey = true,
        FileName = "key.txt"
    },

    SnowEffect = { Enabled = true, ParticleCount = 50, Speed = 1, Glow = true }
})

Tab

Top-level navigation tabs. Pass an icon name from the built-in Lucide icon set.

Lua
local MainTab     = Window:CreateTab("Main",     "home")
local SettingsTab = Window:CreateTab("Settings", "settings")

Sub-Tab

Nested tabs within a main tab for deeper organisation.

Lua
local CombatTab = Window:CreateTab("Combat", "sword")
local SubTabs   = CombatTab:CreateSubTab({"Aimlock", "Camlock", "Triggerbot"})

SubTabs["Aimlock"]:CreateSection("Aimlock Settings")
SubTabs["Aimlock"]:CreateToggle({
    Name = "Enable Aimlock",
    CurrentValue = false,
    Callback = function(Value) print("Aimlock:", Value) end
})

Section

A labelled divider to group related elements.

Welcome
Lua
MainTab:CreateSection("Welcome")

Button

Clickable / tappable button that fires a callback.

Lua
MainTab:CreateButton({
    Name = "Click Me",
    Callback = function() print("Button clicked!") end
})

Toggle

On / off switch. Returns its state to a callback.

Enable Feature
Lua
MainTab:CreateToggle({
    Name = "Enable Feature",
    CurrentValue = false,
    Flag = "FeatureEnabled",
    Callback = function(Value) print("Toggle:", Value) end
})

Slider

Value picker with configurable min / max. Drag with mouse or finger.

Speed50
Lua
MainTab:CreateSlider({
    Name = "Speed",
    Min = 0, Max = 100, CurrentValue = 50,
    Callback = function(Value) print("Speed:", Value) end
})

TextBox

Text input field. Fires callback on change and optionally on Enter.

Username
Lua

    Name = "Username",
    Placeholder = "Enter text...",
    CurrentValue = "",

})

Label

Dynamic text that can be updated programmatically.

Hello World!
Lua

MyLabel:SetText("Updated text!")

Paragraph

Title + description block for informational text.

Information

This is a longer description text.

Lua

    Title = "Information",

})

Toggle Feature

Accent Color

Notification

Toast notification. Appears in-game for the specified duration.

!
Success!

Operation completed successfully.

Lua
Window:Notify({
    Title = "Success!",
    Content = "Operation completed successfully.",
    Duration = 3
})

Invisible Button

A draggable floating button — ideal for mobile actions. Invisible when UI is closed.

Lua — Basic
local Btn = Window:CreateInvisibleButton({ Size = 50, TapCount = 3, SavePosition = true })
Lua — With Image & Callback
local Btn = Window:CreateInvisibleButton({
    Size = 50, TapCount = 3, SavePosition = true,
    Image = "rbxassetid://123456789",
    Callback = function() Window:Toggle() end
})
Btn:SetImage("star"); Btn:SetSize(60); Btn:SetTapCount(2)

Default Settings Tab

Auto-generates a complete Settings tab — theme picker, scale slider, size presets, and version info.

Lua — One-liner
local SettingsTab = Window:CreateDefaultSettingsTab()
Lua — Custom Options
local SettingsTab = Window:CreateDefaultSettingsTab({
    TabName = "Settings", TabIcon = "settings",
    ShowThemeCustomization = true, ShowUIScale = true,
    ShowUISizePresets = true, ShowNotifications = true,
    ShowControls = true, ShowInfo = true,
    Version = "1.0.0", AboutText = "My custom script.",
    CustomThemePresets = {
        ["Ocean"] = Color3.fromRGB(6, 182, 212),
        ["Forest"] = Color3.fromRGB(34, 197, 94)
    }
})

Example Script

Full universal script — PC & Mobile compatible

EliteUI v3.7.0 — Universal Script
-- EliteUI v3.7.0 Universal Script
-- PC & Mobile compatible

local EliteUI = loadstring(game:HttpGet("https://raw.githubusercontent.com/JJ-key/EliteUI/main/roblox/Loader"))()

local Players          = game:GetService("Players")
local RunService       = game:GetService("RunService")
local UserInputService = game:GetService("UserInputService")
local Workspace        = game:GetService("Workspace")
local Lighting         = game:GetService("Lighting")
local TeleportService  = game:GetService("TeleportService")

local LocalPlayer = Players.LocalPlayer
local Camera      = Workspace.CurrentCamera

local movementEnabled = false; local flyEnabled = false; local noclipEnabled = false
local antiAfkEnabled = true;   local infiniteJumpEnabled = false
local sprintEnabled = false;   local antiVoidEnabled = false; local aimlockEnabled = false
local flySpeed = 50; local walkSpeed = 16; local jumpPower = 50; local sprintMultiplier = 2
local aimlockTarget = nil; local aimlockSmoothness = 50
local connections = {}; local lastSafePosition = nil; local antiVoidDebounce = false

local Window = EliteUI:CreateWindow({
    Name = "Universal Script", LoadingTitle = "Loading...", Subtitle = "v3.7.0",
    ToggleKey = Enum.KeyCode.RightShift,
    SnowEffect = { Enabled = true, ParticleCount = 30, Speed = 1, Glow = true },
    KeySystem = { Enabled = false },
    UIScale = { PC = 1.0, Mobile = 0.85, Tablet = 0.9 },
    Theme = {
        Primary = Color3.fromRGB(88, 101, 242), Background = Color3.fromRGB(18, 18, 24),
        Secondary = Color3.fromRGB(32, 34, 44), Text = Color3.fromRGB(255, 255, 255),
        TextSecondary = Color3.fromRGB(148, 155, 164), Accent = Color3.fromRGB(114, 137, 218)
    }
})

local function getCharacter() return LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait() end
local function getHumanoid() local c = getCharacter(); return c and c:FindFirstChildOfClass("Humanoid") end
local function getRootPart() local c = getCharacter(); return c and (c:FindFirstChild("HumanoidRootPart") or c:FindFirstChild("Torso")) end
local function notify(t, m, d) Window:Notify({ Title=t, Content=m, Duration=d or 3 }) end
local function getNearestPlayer()
    local nearest, dist = nil, math.huge
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character then
            local r = p.Character:FindFirstChild("HumanoidRootPart")
            local h = p.Character:FindFirstChildOfClass("Humanoid")
            if r and h and h.Health > 0 then
                local d = (r.Position - Camera.CFrame.Position).Magnitude
                if d < dist then dist = d; nearest = p end
            end
        end
    end
    return nearest
end
